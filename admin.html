(async function connectSanadMarketingToSupabase() {
  const SUPABASE_URL = "https://eqpfbvsdwzisnecmpoio.supabase.co";
  const SUPABASE_KEY = "sb_publishable_X282wxPhOgxtI9Fht44-nw_L0o8qEXc";
  try {
    const res = await fetch(SUPABASE_URL + "/rest/v1/sanad_config?id=eq.main_config&select=*", {
      headers: { "apikey": SUPABASE_KEY }
    });
    if (!res.ok) return;
    const rows = await res.json();
    if (!rows || !rows[0]) return;
    const data = rows[0];
    const m = (data.plans && data.plans.marketing) ? data.plans.marketing : (data.marketing || {});
    const plansList = Array.isArray(data.plans) ? data.plans : (data.plans && data.plans.items ? data.plans.items : []);
    const postsList = Array.isArray(data.posts) ? data.posts : [];

    // 1. ربط أزرار التنزيل الـ 5 (تبقى فارغة حتى يملأها المدير من بوابة الإدارة)
    const bindBtn = (id, url) => {
      const btn = document.getElementById(id);
      if (!btn) return;
      if (url && url.trim() !== "") {
        btn.href = url.trim();
        btn.setAttribute("data-ready", "true");
      } else {
        btn.href = "javascript:void(0)";
        btn.setAttribute("data-ready", "empty");
      }
    };
    bindBtn("sanad-btn-android", m.downloadAndroid);
    bindBtn("sanad-btn-windows", m.downloadWindows);
    bindBtn("sanad-btn-macos", m.downloadMacos);
    bindBtn("sanad-btn-ios", m.downloadIos);
    bindBtn("sanad-btn-web", m.downloadWeb);

    // 2. تحديث العنوان الرئيسي والفيديو الترويجي
    const heroEl = document.getElementById("sanad-hero-title");
    if (heroEl && m.heroTitle) heroEl.textContent = m.heroTitle;
    const vidEl = document.getElementById("sanad-video-frame");
    if (vidEl && m.promoVideoUrl) vidEl.src = m.promoVideoUrl;

    // 3. عرض الإعلانات التسويقية تلقائياً
    const postsBox = document.getElementById("sanad-posts-box");
    if (postsBox && postsList.length > 0) {
      postsBox.innerHTML = postsList.map(p =>
        '<div class="sanad-post-item"><strong>[' + p.badge + '] ' + p.title + '</strong><p>' + p.body + '</p></div>'
      ).join("");
    }
  } catch (err) {
    console.warn("Sanad Cloud Sync:", err);
  }
})();

-- شغّل هذا الكود مرة واحدة في SQL Editor داخل حسابك على Supabase
CREATE TABLE IF NOT EXISTS public.sanad_accounts (
  email TEXT PRIMARY KEY,
  company TEXT NOT NULL DEFAULT '',
  phone TEXT NOT NULL DEFAULT '',
  device_id TEXT NOT NULL DEFAULT '',
  plan TEXT NOT NULL DEFAULT 'TRIAL',
  plan_no INTEGER NOT NULL DEFAULT 1,
  status TEXT NOT NULL DEFAULT 'ACTIVE',
  days INTEGER NOT NULL DEFAULT 365,
  license_key TEXT NOT NULL DEFAULT '',
  updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS public.sanad_config (
  id TEXT PRIMARY KEY,
  plans JSONB NOT NULL DEFAULT '[]'::jsonb,
  posts JSONB NOT NULL DEFAULT '[]'::jsonb,
  marketing JSONB NOT NULL DEFAULT '{}'::jsonb,
  admin_auth JSONB NOT NULL DEFAULT '{}'::jsonb,
  updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

ALTER TABLE public.sanad_accounts ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.sanad_config ENABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS "Allow public access accounts" ON public.sanad_accounts;
CREATE POLICY "Allow public access accounts" ON public.sanad_accounts FOR ALL USING (true) WITH CHECK (true);

DROP POLICY IF EXISTS "Allow public access config" ON public.sanad_config;
CREATE POLICY "Allow public access config" ON public.sanad_config FOR ALL USING (true) WITH CHECK (true);
